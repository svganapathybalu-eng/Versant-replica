      Cross-Device Scholar Score Collection (Google Sheets Webhook)

      When scholars take the test from their own laptops or phones, connecting a free Google Sheet Web App URL automatically records every scholar's name, TIN, sub-scores, and item-by-item check sheet into your central sheet and pulls them into your Trainer Scoreboard above.

      Google Apps Script (Paste into Extensions → Apps Script in any Google Sheet) 
      function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Date", "Name", "TIN", "Mode", "Overall", "CEFR", "Speaking", "Listening", "Reading", "Writing", "AccuracyPct", "ItemsJSON"]);
  }
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([data.date, data.name, data.tin, data.mode, data.overall, data.cefr, data.speaking, data.listening, data.reading, data.writing, data.accuracyPct, JSON.stringify(data.items || [])]);
  return ContentService.createTextOutput(JSON.stringify({status: "ok"})).setMimeType(ContentService.MimeType.JSON);
}

function doGet() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var out = [];
  for (var i = 1; i < rows.length; i++) {
    var r = rows[i];
    out.push({date: r[0], name: r[1], tin: r[2], mode: r[3], overall: Number(r[4]), cefr: r[5], speaking: Number(r[6]), listening: Number(r[7]), reading: Number(r[8]), writing: Number(r[9]), accuracyPct: Number(r[10]), items: JSON.parse(r[11] || "[]")});
  }
  return ContentService.createTextOutput(JSON.stringify(out.reverse())).setMimeType(ContentService.MimeType.JSON);
}
